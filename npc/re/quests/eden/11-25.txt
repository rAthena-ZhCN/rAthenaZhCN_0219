moc_para01,36,38,3	script	任务 [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- 你需要持有 -";
		mes "- ^4d4dff'乐园团徽章'^000000 -";
		mes "- 才能接取这些任务。 -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes "- 你必须先领取你的 -";
			mes "- 报酬才能开始 -";
			mes "- 新的任务。 -";
			mes " ";
			mes "- 经理：斯派克 -";
			close;
		}
		else if (.@j >= 0) {
			mes "- 你一次只能申请 -";
			mes "- 一项任务。 -";
			mes " ";
			mes "- 经理：斯派克 -";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes "- 目前没有 -";
		mes "- 可接取的任务。 -";
		close;
	}
	if (BaseLevel > 25) {
		mes "- 这些任务对你来说 -";
		mes "- 太简单了 -";
		mes "- 去找更符合 -";
		mes "- 等级的任务吧。 -";
		close;
	}
	mes "- 任务公告板 -";
	mes "- 新手专用。 -";
	mes " ";
	mes "- 经理：斯派克 -";
	next;
	switch(select("蜂兵狩猎:秃鹰狩猎:蝗虫后腿:蚯蚓尾巴狩猎:蘑菇狩猎:害虫驱除:摩卡狩猎:吸血蝙蝠狩猎:收集柔毛:收集毒魔菇孢子:取消")) {
		case 1: callsub L_Quest,11114,"蜂兵猎狩"," - 狩猎10只^4d4dff'蜂兵'^000000 - "," - 普隆德拉西北方。 - ";
		case 2: callsub L_Quest,11115,"秃鹰猎狩"," - 狩猎20只^4d4dff'秃鹰'^000000 - "," - 梦罗克南方。 - ";
		case 3: callsub L_Quest,11116,"^i[940]蝗虫后腿"," - 狩猎10只^4d4dff'蝗虫'^000000 - "," - 普隆德拉西南方，以及 - "," - 带来10个^4d4dff^i[940]蝗虫后腿^000000。 - ";
		case 4: callsub L_Quest,11117,"森灵猎狩"," - 狩猎20只^4d4dff'森灵'^000000 - "," - 裴扬东南方。- ";
		case 5: callsub L_Quest,11118,"魔菇猎狩"," - 狩猎30只^4d4dff'魔菇'^000000 - "," - 裴扬附近。 - ";
		case 6: callsub L_Quest,11119,"害虫驱除"," - 狩猎20只^4d4dff盗虫卵^000000 - "," - 普隆德拉地下水道内 - "," - 并带来10个^4d4dff^i[915]蛹壳^000000。 - ";
		case 7: callsub L_Quest,11120,"摩卡猎狩"," - 狩猎20只^4d4dff'摩卡'^000000 - "," - 普隆德拉南方。 - ";
		case 8: callsub L_Quest,11121,"狸猫猎狩"," - 狩猎20只^4d4dff狸猫^000000 - "," - 地下水道内 - "," - 普隆德拉西方。 - ";
		case 9: callsub L_Quest,11122,"收集羽毛"," - 狩猎30只^4d4dff'七彩大嘴鸟'^000000 - "," - 普隆德拉南方。 - ";
		case 10: callsub L_Quest,11123,"收集^i[7033]毒魔菇芽孢"," - 狩猎30只^4d4dff毒魔菇^000000 - "," - 吉芬塔内，以及 - "," - 带来5个^4d4dff^i[7033]毒魔菇芽孢^000000。 - ";
		case 11:
			mes "- 停止阅读公告板。 -";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes "- 任务 ^4d4dff"+ getarg(1) +"^000000 -";
		mes "- 今天已经完成了。 -";
		mes " ";
		mes "- 经理：斯派克 -";
		close;
	}
	mes "- 任务 '"+ getarg(1) +"' -";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes " ";
	mes "- 经理：斯派克 -";
	next;
	if (select("我要接这个任务。:寻找其他任务。") == 2) {
		mes "- 我们去找找 -";
		mes "- 其他任务吧。 -";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	斯派克	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"蜂兵猎狩",900;
				case 11115: callsub L_Quest,.@i,"秃鹰猎狩",1800;
				case 11116: callsub L_Quest,.@i,"^i[940]蝗虫后腿",1400,940,10;
				case 11117: callsub L_Quest,.@i,"森灵猎狩",2600;
				case 11118: callsub L_Quest,.@i,"魔菇猎狩",3900;
				case 11119: callsub L_Quest,.@i,"消灭害虫",2900,915,10;
				case 11120: callsub L_Quest,.@i,"摩卡猎狩",3200;
				case 11121: callsub L_Quest,.@i,"狸猫猎狩",3200;
				case 11122: callsub L_Quest,.@i,"收集羽毛",5400;
				case 11123: callsub L_Quest,.@i,"收集^i[7033]毒魔菇芽孢",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[斯派克]";
			mes "你还没有完成你的任务。";
			mes "继续加油，你马上就完成了。";
			close;
		}
	}
	mes "[斯派克]";
	mes "你在找工作吗？";
	mes "我们总是有很多工作要做。";
	next;
	mes "[斯派克]";
	mes "不如四处看看，找些有趣的任务吧？";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[斯派克]";
				mes "你没有足够的"+ getitemname(.@item_req) +".";
				mes "你需要带来"+ .@req_amount +" "+ getitemname(.@item_req) +".";
				close;
			}
		}
		mes "[斯派克]";
		mes "嗯……让我看看~";
		mes "你完成了 '"+ getarg(1) +"' 任务。";
		next;
		mes "[斯派克]";
		mes "干得好~！！";
		mes "我们的委托人对你的工作非常满意。他会感谢你所做的一切。";
		mes "请再帮帮我们。谢谢！";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}

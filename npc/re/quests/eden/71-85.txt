moc_para01,44,38,3	script	任务 [71 - 85]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- 仅限 -";
		mes "- 乐园团成员 -";
		mes "- 阅读此公告板。 -";
		close;
	}

	// Clear the original quest data (no longer used).
	if (slv_quest) set slv_quest,0;
	if (checkquest(10102) > -1 || checkquest(10103) > -1 || checkquest(10104) > -1 || checkquest(10105) > -1 || checkquest(10106) > -1) {
		for(set .@quest,10102; .@quest<=10106; set .@quest,.@quest+1) {
			if (checkquest(.@quest) > -1)
				erasequest .@quest;
		}
		mes "- 由于委托人已搬离 -";
		mes "- 到别处， -";
		mes "- 申请表已被撤下。 -";
		next;
	}

	if (BaseLevel < 71) {
		mes "- 仅限 -";
		mes "- 符合等级者 -";
		mes "- 阅读此公告板。 -";
		mes "- 请寻找 -";
		mes "- 适合你等级的公告板。 -";
		close;
	} else if (BaseLevel > 85) {
		// Quest IDs: 10107-10123, 5055-5057
		setarray .@quests[0],
			10107,10108,10109,
			10110,10111,10112,10113,
			10114,10115,10116,10117,
			10118,10119,10120,
			10121,10122,10123,5055,5056;
		for(set .@i,0; .@i<getarraysize(.@quests); set .@i,.@i+1) {
			if (checkquest(.@quests[.@i],HUNTING) == 2)
				set .@complete[getarraysize(.@complete)], .@quests[.@i];
		}
		if (checkquest(5057) > -1 && countitem(7187) >= 30)
			set .@complete[getarraysize(.@complete)],5057;
		if (getarraysize(.@complete)) {
			mes "你有进行中的任务。";
			mes "要提交吗？";
			next;
			if(select("当然:不") == 2)
				close;
			for(set .@i,0; .@i<getarraysize(.@complete); set .@i,.@i+1)
				callsub L_Quest,.@complete[.@i];
			close;
		}
		mes "- 仅限 -";
		mes "- 符合等级者 -";
		mes "- 阅读此公告板。 -";
		mes "- 请寻找 -";
		mes "- 适合你等级的公告板。 -";
		close;
	}
	mes "- 选择你想 -";
	mes "- 狩猎的地方。 -";
	next;
	switch(select("史芬克斯地下城:克雷斯特汉姆古城:朱诺地区:钟塔:本地岛屿")) {
	case 1:
		mes "- 这里有几项 -";
		mes "- 与史芬克斯地下城 -";
		mes "- 相关的委托。 -";
		next;
		set .@str1$,"梦罗克的村民们因史芬克斯地下城里不断增多的魔物而忧心忡忡。 ";
		set .@str2$,"- 梦罗克的梅迪 -";
		switch(select("猎杀诺可伊:猎杀马尔杜克:猎杀帕莎纳")) {
			case 1: callsub L_Quest,10107,.@str1$+"猎杀10只诺可伊。",.@str2$;
			case 2: callsub L_Quest,10108,.@str1$+"猎杀10只马尔杜克。",.@str2$;
			case 3: callsub L_Quest,10109,.@str1$+"猎杀10只帕莎纳。",.@str2$;
		}
	case 2:
		mes "- 这里有几项 -";
		mes "- 与克雷斯特汉姆古城 -";
		mes "- 相关的委托。 -";
		next;
		set .@str1$,"我很害怕克雷斯特汉姆古城的魔物会来袭击我们。 ";
		set .@str2$,"- 吉芬的安塞莉亚 -";
		switch(select("猎杀深渊镜妖:猎杀暗神官:猎杀恶灵:猎杀幽灵弓箭手")) {
			case 1: callsub L_Quest,10110,.@str1$+"猎杀10只镜妖。",.@str2$;
			case 2: callsub L_Quest,10111,.@str1$+"猎杀10只暗神官。",.@str2$;
			case 3: callsub L_Quest,10112,.@str1$+"猎杀10只恶灵。",.@str2$;
			case 4: callsub L_Quest,10113,.@str1$+"猎杀10只幽灵弓箭手。",.@str2$;
		}
	case 3:
		mes "- 这里有几项 -";
		mes "- 与朱诺周边地区 -";
		mes "- 相关的委托。 -";
		next;
		set .@str1$,"像我这样不会战斗的人，因为魔物数量太多，出行变得困难许多。 ";
		set .@str2$,"- 旅行者让拜 -";
		switch(select("猎杀七彩大嘴鸟:猎杀沙妖:猎杀魔羌:猎杀鸟人哈比")) {
			case 1: callsub L_Quest,10114,.@str1$+"猎杀20只七彩大嘴鸟。",.@str2$;
			case 2: callsub L_Quest,10115,.@str1$+"猎杀20只沙妖。",.@str2$;
			case 3: callsub L_Quest,10116,.@str1$+"猎杀20只魔羌。",.@str2$;
			case 4: callsub L_Quest,10117,.@str1$+"猎杀20只鸟人哈比。",.@str2$;
		}
	case 4:
		mes "- 这里有几项 -";
		mes "- 与钟塔 -";
		mes "- 相关的委托。 -";
		next;
		set .@str1$,"我以前很喜欢聆听艾尔帕兰钟塔那悦耳的旋律，但现在魔物吵闹不停，我什么都听不见了。 ";
		set .@str2$,"- 钟表同好会会长里辛塞特 -";
		switch(select("猎杀古钟魔:猎杀朽魔:猎杀飞行魔书")) {
			case 1: callsub L_Quest,10118,.@str1$+"趁钟塔还没倒塌，猎杀15只古钟魔。",.@str2$;
			case 2: callsub L_Quest,10119,.@str1$+"趁钟塔还没倒塌，猎杀15只朽魔。",.@str2$;
			case 3: callsub L_Quest,10120,.@str1$+"趁钟塔还没倒塌，猎杀15只飞行魔书。",.@str2$;
		}
	case 5:
		mes "- 这里有几项 -";
		mes "- 与本地岛屿 -";
		mes "- 相关的委托。 -";
		next;
		set .@str1$,"我想去冒险，可手头要处理的事情实在太多了。我若不行，别人也肯定不行！ ";
		set .@str2$,"- 想去冒险的芬菲 -";
		switch(select("猎杀树林女巫:猎杀米雅毕:猎杀米糕:猎杀无头骡:猎杀塔噜安:为庆典收集庆典面具")) {
			case 1: callsub L_Quest,10121,.@str1$+"在莫斯科比亚猎杀15只树林女巫！",.@str2$;
			case 2: callsub L_Quest,10122,.@str1$+"在天津町猎杀15只米雅毕！",.@str2$;
			case 3: callsub L_Quest,10123,.@str1$+"在洛阳猎杀15只米糕！",.@str2$;
			case 4: callsub L_Quest,5055,"我知道城外有许多可怕的魔物，但让我害怕的却是这些在附近徘徊的无头骡。希望它们永远不要闯进城里。猎杀15只无头骡！","- 巴西的玛丽安娜 -";
			case 5: callsub L_Quest,5056,"塔噜安太吵了。早上我还能忍受，可到了晚上它们吵得我睡不着。我想睡觉。猎杀15只塔噜安！","- 哎哟泰雅的萨满普莫 -";
			case 6: callsub L_Quest,5057,"元宵童子把我为庆典准备的所有^i[7187]庆典面具都偷走了。请设法弄到30个面具！这份人情我记下了~","- 昆仑族长萨尤姆 -";
		}
	}
	end;

//callsub L_Quest,<quest ID>{,"<description1>","<description2>"};
L_Quest:
	set .@quest, getarg(0);
	switch(.@quest) {
		case 10107: set .@name$,"猎杀诺可伊"; set .@exp,15000; break;
		case 10108: set .@name$,"猎杀马尔杜克"; set .@exp,15000; break;
		case 10109: set .@name$,"猎杀帕莎纳"; set .@exp,15000; break;
		case 10110: set .@name$,"猎杀镜妖"; set .@exp,20000; break;
		case 10111: set .@name$,"猎杀暗神官"; set .@exp,20000; break;
		case 10112: set .@name$,"猎杀恶灵"; set .@exp,20000; break;
		case 10113: set .@name$,"猎杀幽灵弓箭手"; set .@exp,20000; break;
		case 10114: set .@name$,"猎杀七彩大嘴鸟"; set .@exp,30000; break;
		case 10115: set .@name$,"猎杀沙妖"; set .@exp,30000; break;
		case 10116: set .@name$,"猎杀魔羌"; set .@exp,30000; break;
		case 10117: set .@name$,"猎杀鸟人哈比"; set .@exp,30000; break;
		case 10118: set .@name$,"猎杀古钟魔"; set .@exp,25000; break;
		case 10119: set .@name$,"猎杀朽魔"; set .@exp,25000; break;
		case 10120: set .@name$,"猎杀飞行魔书"; set .@exp,25000; break;
		case 10121: set .@name$,"猎杀树林女巫"; set .@exp,30000; break;
		case 10122: set .@name$,"猎杀米雅毕"; set .@exp,30000; break;
		case 10123: set .@name$,"猎杀米糕"; set .@exp,30000; break;
		case 5055: set .@name$,"猎杀无头骡"; set .@exp,30000; break;
		case 5056: set .@name$,"猎杀塔噜安"; set .@exp,30000; break;
		case 5057: set .@name$,"为庆典收集^i[7187]庆典面具"; set .@exp,30000; break;
	}
	if (getargcount() == 1) {
		mes "'"+.@name$+"'已完成。要提交并领取奖励吗？";
		next;
		if(select("提交任务:取消") == 1) {
			if (.@quest == 5057) delitem 7187,30; //Festival_Mask
			erasequest .@quest;
			getexp .@exp,0;
			mes "完成任务后获得了经验值。";
			next;
			return;
		}
		close;
	} else {
		if (.@quest == 5057) {
			// Quest 5057 is the only collection quest, so separate conditions are unnecessary.
			if (checkquest(5057) == -1)
				set .@hunting,-1;
			else if (countitem(7187) >= 30)
				set .@hunting,2;
		} else
			set .@hunting, checkquest(.@quest,HUNTING);
		if (.@hunting == 0 || .@hunting == 1) {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "完成任务后请回到这里。";
		} else if (.@hunting == 2) {
			mes "你已完成任务。";
			next;
			if(select("提交任务:取消") == 1) {
				if (.@quest == 5057) delitem 7187,30; //Festival_Mask
				erasequest .@quest;
				getexp .@exp,0;
				mes "完成任务后获得了经验值。";
			}
		}/* else if (.@hunting == 3) {
//			recall_completequest .@quest
			erasequest .@quest;
			mes "你已重新接取任务 '"+.@name$+"'.";
		}*/ else {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "要接受任务吗？";
			next;
			if(select("是。:否。") == 1) {
				mes "已接取任务 '"+.@name$+"'。更多详情请查看任务信息。";
				setquest .@quest;
			}
		}
		close;
	}
}

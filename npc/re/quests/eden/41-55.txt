			close;
		}
		mes "[任务公告板]";
		mes "任务公告板。";
		next;

			mes "^000077好吧。下次再查看任务。^000000。";

		}

			case 3: callsub L_Quest,12074,"有人在海滩附近被抢了！嫌疑犯是名叫库克雷的犯罪魔物。请狩猎30只库克雷。";
			case 4: callsub L_Quest,12075,"最近兽人婴儿越来越暴躁了。请狩猎15只兽人婴儿。";
			case 5: callsub L_Quest,12076,"兽人战士的攻击性越来越强了。请狩猎20只兽人战士。";

			case 8: callsub L_Quest,12079,"你能去狩猎乌贼吗？我的研究需要15只乌贼。请帮帮我。";

			erasequest 12072+.@i;

		close;

		mes "[任务公告板]";
		mes "你已接受任务。更多详情请按 'alt + u' 键，在任务窗口中查看信息。";
		setquest 12088;
		setquest getarg(0);
